########## NAME
M_lactate5

########## METADATA
time_unit = m
description = """

This model describes the Epinephrine and Norepinephrine dynamics during excercise.
Epinephrine (EPI),
Norepinephrine (NOR),
Percentage of maximal oxygen consumption (PVo2max)

o2 deficit met push term 
"""
########## MACROS
########## STATES
d/dt(PVo2max) = -0.8*PVo2max + 0.8*u3                                                                       # Percentage of maximal oxygen consumption (dimensionless)
d/dt(O2_deficit) = v_cons * f_hill(max(PVo2max-threshold,0), km_O2, n_O2) - k_restore * O2_deficit + (k_push *  max(time - time_ex, 0) + k_push2) * PVo2max* sign(u3)                                   # - consuption + restore
d/dt(lactate) = vmax*f_hill(O2_deficit, km_lac, n_lac) +  - elim_lactate*(lactate-Lacb) 

PVo2max(0) = 0.0    # dimensionless
lactate(0) = 1.0    # mmol/L
O2_deficit(0) = 0.0  # %

########## PARAMETERS
# O2_deficit
v_cons = 0.64
km_O2 = 9.33
n_O2 = 1.037
k_restore = 10.0

k_push = 1.0
k_push2 = 1.0

# Lactate
vmax = 7.5
km_lac = 0.118
n_lac = 1.9
elim_lactate = 0.073

# Aerobic threshold
vmax_thresh = 100.0
km_thresh = 0.68
n_thresh = 1.0

########## VARIABLES
threshold = vmax_thresh * f_hill(Vo2max/90, km_thresh, n_thresh)    # Aerobic threshold

########## FUNCTIONS
f_hill(x, km, n) = x^n / (km^n + x^n)
f_mm (x, km) = x / (km + x) 

########## EVENTS
event_basal = gt(time, -INFINITY), lactate, Lacb, O2_deficit, O2b

########## OUTPUTS
########## INPUTS
u3 = intensity  @ 0          # exercise intensity (%)
Ib = Ib         @ 12.0       # basal insulin level (mU/L)
Gb = Gb         @ 4.9750     # basal plasma glucose (mmol/L)
Epib = Epib         @ 0.3     # basal epinephrine (nmol/)
Norb = Norb         @ 1.3     # basal norepinephrine (nmol/L)
Lacb = Lacb         @ 1.0     # basal Lactate (mmol/L)
BW = BW           @ 84         # body weight (kg)
Vo2max = Vo2max   @ 40        # percentage of maximal oxygen consumption
time_ex = time_ex @ 0
O2b = O2b         @ 0.0

########## FEATURES
PVo2max = PVo2max
Lactate_mmolL = lactate
lactate = lactate
O2_deficit = O2_deficit
