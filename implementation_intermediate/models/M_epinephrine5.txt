########## NAME
M_epinephrine5

########## METADATA
time_unit = m
description = """

This model describes the Epinephrine and Norepinephrine dynamics during excercise.
Epinephrine (EPI),
Norepinephrine (NOR),
Percentage of maximal oxygen consumption (PVo2max)

old PVO2max equation
lactate added
"""
########## MACROS
########## STATES
d/dt(PVo2max) = -0.8*PVo2max + 0.8*u3                                                                             #  percentage of maximal oxygen consumption (dimensionless)

d/dt(EPI)    = prod_epi*stress - elim_epi* (EPI-basal_epi)                                                        # Epinephrine (nmol/L/min)
d/dt(NOR)    = prod_nor*stress - elim_nor*(NOR - basal_nor)                                                       # Norepinephrine (nmol/L/min)
d/dt(stress) = ((Vmax_stress *(PVo2max)^n_stress)/(Km_stress^n_stress + PVo2max^n_stress)) - elim_stress*stress + lac_stress * lactate
d/dt(lactate) = prod_lactate*max(PVo2max-LT1,0) - elim_lactate*lactate

PVo2max(0) = 0.0 # dimensionless
EPI(0) = 0.3
NOR(0) = 1.3
stress(0) = 0
lactate(0) = 0

########## PARAMETERS
# Epi
prod_epi = 0.1  # 0.559   
elim_epi = 10   # 113      
basal_epi = 0.3 # 0.28       

# Nor
prod_nor = 0.3  # 19  
elim_nor = 10   # 490    
basal_nor = 1.3 # 1.2886       

# Stress
Vmax_stress = 6.0  # 20         
Km_stress = 40     # 1.68        
n_stress = 1       # 1.0 
elim_stress = 0.05 # 0.188   
lac_stress = 0.1   # 0.139 

# Lactate
prod_lactate = 0.05 # 1751 
elim_lactate = 0.5  # 84
LT1 = 50            # 52

########## VARIABLES
########## FUNCTIONS
########## EVENTS
########## OUTPUTS
########## INPUTS
u3 = intensity  @ 0          # exercise intensity (%)
Ib = Ib         @ 12.0       # basal insulin level (mU/L)
Gb = Gb         @ 4.9750     # basal plasma glucose (mmol/L)
BW = BW         @ 84         # body weight (kg)
#Vo2max = Vo2max @ 40        # percentage of maximal oxygen consumption

########## FEATURES
PVo2max = PVo2max
Norepinephrine_nmolL = NOR
Epinephrine_nmolL = EPI
EPI = EPI
NOR = NOR
stress = stress
lactate = lactate
