########## NAME
M_parker_low_intensities

########## METADATA
time_unit = m
description = """

This model describes the glucose and insulin dynamics during excercise
Plamsa insulin (I),
Remote Insulin (X),
Plasma glucose (G),

First model
"""
########## MACROS
########## STATES
d/dt(I)     = -n*I + p4*(u1+u1b) - Ie # mU/L/min
d/dt(X)     = -p2*X + p3*(I-((p4/n)*u1b))  # mU/L/min
d/dt(G)     = -p1*(G - Gb) - X*G + (BW/VolG)*(Gprod - Ggly) - (BW/VolG)*Gup + u2/VolG # mmol/L/min
d/dt(Gprod) = a1*PVo2max - a2*Gprod    # hepatic glucose production induced by exercise (mmol/min)
d/dt(Gup)   = a3*PVo2max - a4*Gup      # glucose uptake induced by excercise (mmol/min)
d/dt(Ie)    = a5*PVo2max - a6*Ie       # rate of insulin removal (mU/L/min)
d/dt(PVo2max) = -0.8*PVo2max + 0.8*u3  # 
d/dt(Ggly) = gly_A*k*A_u3 - Ggly/T1*B_u3 # glycolyse dynamics (simplified for now) (mmol/kg/min)
d/dt(A)    = A_u3*u3 - (A/0.001)*B_u3   # integrated exercise intensity (simpLified)

I(0)     = 12.0   # mU/L
X(0)     = 0.0  # mU/L
G(0)     = 4.9750   # mmol
Gprod(0) = 0.0  # mmol/kg
Gup(0)   = 0.0  # mmol/kg
Ie(0)    = 0.0  # (mU/L/min)
PVo2max(0) = 0.0
Ggly(0) = 0.0   # mmol
A(0)    = 0.0

########## PARAMETERS
n   = 0.142       # clearance rate plasma insuline (min^-1)
p1  = 0.035        # glucose removal rate from plasma space (independent of insulin) (min^-1)
p2  = 0.05         # disappearance rate remote insulin from remote insulin compartment (min^-1)
p3  = 0.000028     # appearance rate insulin in remote insulin compartment (L/mU*min^2)
p4  = 98.0         # (L^-1)
VolG = 11.7        # glucose distribution space (L)
a1  = 0.000008769982238011767  # (mmol*kg*min^2)
a2  = 0.056             # (min^-1)
a3  = 0.000010823712255772646  # (mmol*kg*min^2)
a4  = 0.0485            # (min^-1)
a5  = 0.00125             # (mU*L^-1*min)
a6  = 0.075        #  (min^-1)
k   = 0.0000599467 #0.0108  #  rate of glycolyses decline (mmol*kg^-1*min^2)
T1  = 6.0          #  time required o return to basel glycolyses value (min)
 
########## VARIABLES
ATH = (-1.1521*u3^2)+(87.471*u3)    # critical threshold value at which glycolyses start to decrease during excercise (%-min) 
gly_A = (sign(A-ATH+1)+1)/2
A_u3 = 1.0*sign(u3)
B_u3 = (sign(1-u3)+1)/2
u1b = Ib*n/p4
u1 = 0         # insulin infused exogenously (mU/min)
u2 = 0         # dietary absorption or external infusion of glucose (mmol/min)


########## FUNCTIONS
########## EVENTS
########## OUTPUTS
########## INPUTS
u3 = intensity  @ 0          # exercise intensity
Ib = Ib         @ 12.0       # basal insulin level (mU/L)
Gb = Gb         @ 4.9750     # basal plasma glucose (mmol/L)
BW = BW         @ 84         # body weight (kg)
#Vo2max = Vo2max @ 40        # percentage of maximal oxygen consumption

########## FEATURES
Glucose_mmolL = G
Glucose_mgdl = G/0.0555
Insulin_mUL = I
