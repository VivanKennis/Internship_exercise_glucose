########## NAME
M_cortisol

########## METADATA
time_unit = m
description = """This model descibes the concentration of stress hormones during exercise"""

########## MACROS

########## STATES
d/dt(PVO2max) = -p1*PVO2max + p2*ex_effect

// Kajsas kortisolmodel  (CRH, ACTH and cortisol are in ugram/dL, the rest has no units)
d/dt(CRH) = p11 + p222*PVO2max/(1+CORT) - d1*CRH + p333*max(0,GTH-Gb) // look at this                          // *circadian
d/dt(ACTH) = p22*CRH/(1+CORT) - d2*ACTH                                                  // mean value pmol/L
d/dt(CORT) = p33*ACTH - d3*CORT                                                         // mean value ug/dL
// d/dt(circadian) = (circ - circadian) * (1/t_circ)

d/dt(mod) = 0

PVO2max(0) = 0

CRH(0) = 2.048 
ACTH(0) = 3.81 
CORT(0) = 8.12 
// circadian(0) = 1


mod(0)=0.0

########## PARAMETERS
// Initial glucose - cortisol params
p111 = 0.4812 // -2.0551            
p333 = 1

p1 = 0.8
p2 = 0.8

// Cortisol model
// All reactions are in ugram/dL/min
p11 = 1.64 
p22 = 0.7200 
p33 = 0.0960 

d1 = 1.6611 
d2 = 0.0376 
d3 = 0.0104 

// exercise 
vmax = 2
Km = 77
h = 3

########## VARIABLES

ex_effect = intensity // (intensity > intensityTH) ? (intensity) : 1
p222 = vmax*ex_effect^h/(Km+ex_effect)^h // 0.0828 // 10.3343

GTH = 59.4595
Ra = 1
Rd = 1
Epi = 1
NE = 1

########## FUNCTIONS

########## EVENTS
event3 = intensity<=0        //do nothing, but we do it to reset the event triggering. 
event4 = intensity>0        //do nothing, but we do it to reset the event triggering. 

########## OUTPUTS

########## INPUTS
VO2max = VO2max     @ 40
intensity = intensity             @ 0 // exercise
Gb = Gb             @ 89.6296 // basal glucose level
Ib = Ib             @ 12.0    // basal insulin level

########## FEATURES

CRH_pmolL = CRH
ACTH_pmolL = ACTH
Cortisol_ugdL = CORT
Cortisol_nmolL = CORT*27.6  // converting from ug/dL to nmol/L (cortisol molar mass = 362.46, 1/362.46 = 27.6)
Glucose_mmolL = Gb
Insulin_mlUL = Ib
EGP_mgkgmin = Ra
Rd_mgkgmin = Rd
Epinephrine_nmolL = Epi
Norepinephrine_nmolL = NE
