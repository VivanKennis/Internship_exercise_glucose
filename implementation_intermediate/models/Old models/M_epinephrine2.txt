########## NAME
M_epinephrine2

########## METADATA
time_unit = m
description = """

This model describes the Epinephrine and Norepinephrine dynamics during excercise.
Epinephrine (EPI),
Norepinephrine (NOR),
Percentage of maximal oxygen consumption (PVo2max)

Palumbo model
"""
########## MACROS
########## STATES
#d/dt(PVo2max) = ((Vmax * u3)/(Km + u3)) - Kel * PVo2max                                 # percentage of maximal oxygen consumption (dimensionless)
#d/dt(PVo2max) = -0.8*PVo2max + 0.8*u3                                                    # dimensionless 
#d/dt(EPI)    = a7*stress - a8* (EPI-Eb)                                                 # Epinephrine (nmol/L/min)

d/dt(PVo2max) = -0.8*PVo2max + 0.8*u3 
d/dt(EPI)    = Eb + (BW/VolG)*f2 + f3 - Krecovery*EPI                                   # Epinephrine (nmol/L/min)

d/dt(stress) = Kstress*PVo2max - Krecoverys*stress
d/dt(NOR)    = a9*stress - a10*(NOR - Nb)                                                # Norepinephrine (nmol/L/min)

d/dt(G) = 0                                                                              # Glucose (mmol/L/min)

stress(0) = 0
PVo2max(0) = 0.0    # dimensionless
EPI(0) = 0.3        # (nmol/L)
NOR(0) = 1.3        # (nmol/L)
G(0) = Gb           # (mmol/L)

########## PARAMETERS
Vmax = 80      # (min^-1) # 0.005926235516433483
Km = 80         # (%)     # 76956 
Kel = 0.8      # (min^-1) # 1248.522729415323

#a7 = 0.1    # 2531.99394223619
#a8 = 10     # 0.24315335106983713

a9 = 0.3    # 59845.51229448876
a10 = 10    # 0.6384429376673523
Eb = 0.3    # 0.23693224740282945 
Nb = 1.3    # 1.167043520210867 

VolG = 11.7        # glucose distribution space (L)

c1 = 0.1    
c2 = 2.0
c3 = 3.5

d1 = 2.0    
d2 = 0.05
d3 = 30.0

Kstress = 0.01  # 3148.274826419986
Krecoverys = 0.05 # 0.1889156328224901

########## VARIABLES
f2 = c1 / (1 + exp(c2 * ( G - c3)))
f3 = d1 / (1 + exp(d2 * ( d3- PVo2max)))

f2_basal = c1 / (1 + exp(c2 * (Gb - c3)))
f3_basal = d1 / (1 + exp(d2 * (d3 - 0)))
Krecovery = (Eb + f2_basal + f3_basal) / 0.3

########## FUNCTIONS
########## EVENTS
########## OUTPUTS
########## INPUTS
u3 = intensity   @ 0          # exercise intensity (%)
Ib = Ib          @ 12.0       # basal insulin level (mU/L)
Gb = Gb          @ 4.9750     # basal plasma glucose (mmol/L)
BW = BW          @ 84         # body weight (kg)
#Vo2max = Vo2max @ 40        # percentage of maximal oxygen consumption

########## FEATURES
PVo2max = PVo2max
Norepinephrine_nmolL = NOR
Epinephrine_nmolL = EPI
EPI = EPI
NOR = NOR
stress = stress
G=G
