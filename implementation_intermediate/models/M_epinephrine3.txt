########## NAME
M_epinephrine3

########## METADATA
time_unit = m
description = """

This model describes the Epinephrine and Norepinephrine dynamics during excercise.
Epinephrine (EPI),
Norepinephrine (NOR),
Percentage of maximal oxygen consumption (PVo2max)

old PVO2max equation
"""
########## MACROS
########## STATES
d/dt(PVo2max) = -0.8*PVo2max + 0.8*u3                                                    # dimensionless 

d/dt(EPI)    = a7*stress - a8* (EPI-Eb)                                                        # Epinephrine (nmol/L/min)
d/dt(NOR)    = a9*stress - a10*(NOR - Nb)                                                       # Norepinephrine (nmol/L/min)
#d/dt(stress) = Kstress*PVo2max - Krecovery*stress
d/dt(stress) = ((Vmaxs *(PVo2max)^ns)/(Kms^ns + PVo2max^ns)) - Krecovery*stress

PVo2max(0) = 0.0 # dimensionless
EPI(0) = 0.3
NOR(0) = 1.3
stress(0) = 0

########## PARAMETERS
# PVO2max
Vmax = 80      # (min^-1) # 29424

# Epi

a8 = 10         # 0.3261149894216939

# Nor
a9 = 0.3        # 8250.992616427175

# Stress
#Kstress = 0.01

Vmaxs = 6.0         # 0.2159921836327553
Kms = 40            # 18487.838146515463
ns = 1              # 1.0

########## FUNCTIONS
########## EVENTS
########## OUTPUTS
