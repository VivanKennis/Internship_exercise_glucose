########## NAME
M_epinephrine9

########## METADATA
time_unit = m
description = """

This model describes the Epinephrine and Norepinephrine dynamics during excercise.
Epinephrine (EPI),
Norepinephrine (NOR),
Percentage of maximal oxygen consumption (PVo2max)

nor as precursor of epi
deleted a
"""
########## MACROS
########## STATES
d/dt(PVo2max) = -0.8*PVo2max + 0.8*u3                                                                       # Percentage of maximal oxygen consumption (dimensionless)
d/dt(NOR)    = prod_nor*stress - elim_nor*(NOR- Norb)  #- to epinephrine                                                      # Norepinephrine (nmol/L/min)
d/dt(EPI)    = (prod_epi- nor_loss)*(NOR- Norb) - elim_epi* (EPI-Epib)   # + to epinephrine                              # Epinephrine (nmol/L/min)
d/dt(stress) = ((Vmax_stress *(max(PVo2max,0))^n_stress)/(Km_stress^n_stress + (max(PVo2max,0))^n_stress)) - elim_stress*stress + lac_stress * lactate
d/dt(lactate) = k_exp * (exp(prod_lactate * PVo2max) - 1) - elim_lactate*lactate

PVo2max(0) = 0.0 # dimensionless
EPI(0) = 0.3
NOR(0) = 1.3
stress(0) = 0
lactate(0) = 0

########## PARAMETERS
# Epi
prod_epi = 0.2     
elim_epi = 80.0
nor_loss = 15         

# Nor
prod_nor = 0.3        
elim_nor = 25        

# Stress
Vmax_stress = 10.0         
Km_stress = 49            
n_stress = 3.0
elim_stress = 0.05   
lac_stress = 3

# Lactate
prod_lactate = 0.06
elim_lactate = 0.5
k_exp = 0.3

########## VARIABLES
########## FUNCTIONS
########## EVENTS
########## OUTPUTS
########## INPUTS
u3 = intensity  @ 0          # exercise intensity (%)
Ib = Ib         @ 12.0       # basal insulin level (mU/L)
Gb = Gb         @ 4.9750     # basal plasma glucose (mmol/L)
Epib = Epib         @ 0.3     # basal epinephrine (nmol/)
Norb = Norb         @ 1.3     # basal norepinephrine (mmol/L)
BW = BW         @ 84         # body weight (kg)
#Vo2max = Vo2max @ 40        # percentage of maximal oxygen consumption

########## FEATURES
PVo2max = PVo2max
Norepinephrine_nmolL = NOR
Epinephrine_nmolL = EPI
EPI = EPI
NOR = NOR
stress = stress
lactate = lactate
